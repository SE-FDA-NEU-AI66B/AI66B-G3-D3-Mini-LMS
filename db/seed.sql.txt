-- ================== QUESTIONS (4 per quiz = 48 total) ===============
INSERT INTO question (quiz_id, position, question_type, prompt, accepted_answers, points)
SELECT
    q.quiz_id,
    g.pos,
    CASE WHEN g.pos IN (1, 2) THEN 'MCQ'::question_type
         ELSE 'SHORT'::question_type END,
    'Question ' || g.pos || ' of ' || q.title,
    CASE WHEN g.pos IN (3, 4) THEN 'sample accepted answer' ELSE NULL END,
    1
FROM quiz q
CROSS JOIN generate_series(1, 4) AS g(pos);

-- ==================== OPTIONS (4 per MCQ question) ==================
INSERT INTO option (question_id, label, text, is_correct)
SELECT
    q.question_id,
    l.label,
    'Option ' || l.label || ' for question ' || q.question_id,
    (l.label = 'A')
FROM question q
CROSS JOIN (VALUES ('A'), ('B'), ('C'), ('D')) AS l(label)
WHERE q.question_type = 'MCQ';
